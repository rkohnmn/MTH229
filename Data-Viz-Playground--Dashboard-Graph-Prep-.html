ball_data <- read_csv("C:/Users/RobTop/Documents/STA227 - Data Visualization/vb/ball_positions.csv")
## Rows: 267082 Columns: 7
## ── Column specification ────────────────────────────────────────────────────────
## Delimiter: ","
## dbl (7): frame, u, v, xmin, ymin, xmax, ymax
## 
## ℹ Use `spec()` to retrieve the full column specification for this data.
## ℹ Specify the column types or set `show_col_types = FALSE` to quiet this message.
ball_data$x_court <- -9 + ball_data$u * 18  # u in [0,1] mapped to [-9,9]
#ball_data$y_court <- ball_data$v * 9       # v in [0,1] mapped to [0,9]
y_min_clip <- 2.9
y_max_clip <- 9
ball_data$y_court <- pmax(pmin(ball_data$v * 9, y_max_clip), y_min_clip)  # clip
ball_data$y_court <- (ball_data$y_court - y_min_clip) / (y_max_clip - y_min_clip) * 9  # stretch to 0-9

z <- kde2d(
  x = ball_data$x_court,
  y = ball_data$y_court,
  n = 200,                  # higher resolution
  lims = c(-9, 9, 0, 9)
)
z$z <- sqrt(sqrt(z$z))#more sensitive

# 3. Set up plot with extra right margin for legend
par(mar = c(2,2,2,6))  # bottom, left, top, right

# 4. Plot heatmap
image(
  z,
  col = hcl.colors(20, "YlOrRd", rev = TRUE),  # more colors for sensitivity
  xlim = c(-9, 9),
  ylim = c(0, 9),
  axes = FALSE,
  xlab = "",
  ylab = "",
  main = "Volleyball Ball Heatmap"
)

# 5. Overlay court boundaries
rect(-9, 0, 9, 9, border = "black", lwd = 2)   # outer boundary
abline(v = 0, lwd = 2)                          # centerline (net)
abline(v = -3, lty = 2)                         # attack lines
abline(v = 3,  lty = 2)

# 6. Add half-court shading
rect(-9, 0, 0, 9, border = NA, col = rgb(0.9, 0.9, 1, 0.025))
rect(0, 0, 9, 9, border = NA, col = rgb(1, 0.9, 0.9, 0.025))

rect(-9, 0, -3, 9, border = NA, col = rgb(1, 1, 1, 0.1))  # Team A attack (dark red, transparent)
rect(-3, 0, 0, 9, border = NA, col = rgb(0.5, 0, 0, 0.1))      # Team A defense (white, transparent)
rect(3, 0, 9, 9, border = NA, col = rgb(1, 1, 1, 0.1))      # Team B attack (white, transparent)
rect(0, 0, 3, 9, border = NA, col = rgb(0, 0.5, 0, 0.1))    # Team B defense (dark green, transparent)


# 8. Optional team labels
text(-4.5, 8.5, "Poland", cex = 0.9)
text( 4.5, 8.5, "Italy", cex = 0.9)

# 9. Legend outside the court
cols <- hcl.colors(20, "YlOrRd", rev = TRUE)
legend_x <- 9.2
legend_y <- seq(0, 9, length.out = length(cols)+1)
for(i in 1:length(cols)){
  rect(legend_x, legend_y[i], legend_x + 0.5, legend_y[i+1], col = cols[i], border = NA)
}
# Add numerical labels for density
text(legend_x + 0.6, seq(0, 9, length.out=5), 
     labels = round(seq(min(z$z), max(z$z), length.out=5), 2), adj = 0)
